const cities = [
{ name: "beirut", timeZone: "Asia/Beirut" },
{ name: "dubai", timeZone: "Asia/Dubai" },
{ name: "riyadh", timeZone: "Asia/Riyadh" },
{ name: "singapore", timeZone: "Asia/Singapore" },
{ name: "barcelona", timeZone: "Europe/Madrid" }
];

function updateTimes() {
if (window.innerWidth <= 992) {
return;
}

const now = new Date();
const times = cities.map(city => {
const time = now.toLocaleTimeString("en-US", {
timeZone: city.timeZone,
hour: "2-digit",
minute: "2-digit",
second: "2-digit",
hour12: false,
});
return {
name: city.name,
formattedTime: time.split(':').map(t => t.trim())
};
});

times.forEach(({ name, formattedTime }) => {
const [hours, minutes, seconds] = formattedTime;

document.querySelector(`.time__hours.${name}`).innerHTML = hours;
document.querySelector(`.time__minutes.${name}`).innerHTML = minutes;
document.querySelector(`.time__seconds.${name}`).innerHTML = seconds;

const deg = 6;
const clockHours = (hours % 12) * 30;
const clockMinutes = minutes * deg;

document.querySelector(`.clock__hours.${name}`).style.transform = `rotateZ(${clockHours + (clockMinutes / 12)}deg)`;
document.querySelector(`.clock__minutes.${name}`).style.transform = `rotateZ(${clockMinutes}deg)`;
});
}

setInterval(updateTimes, 1000);
updateTimes();
